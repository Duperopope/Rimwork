    public void RegenerateResources()
    {
        // Placeholder for resource regeneration logic
        foreach (var sapling in _saplings.ToList())
        {
            if (sapling.TicksRemaining <= 0)
            {
                // Regrow the sapling into a tree
                _resources.Add(new ResourceNode(ResourceKind.Tree, sapling.X, sapling.Y));
                _saplings.Remove(sapling);
            }
            else
            {

        }
