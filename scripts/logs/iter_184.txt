private void SpawnPawns(string[] names)
{
    foreach (string name in names)
    {
        int x;
        int y;
        do
        {
            x = _rng.Next(0, _world.Map.Width);
            y = _rng.Next(0, _world.Map.Height);
        }
        while (!_world.Map.IsPassable(x, y));
        _world.RegisterThing(new Pawn(name, x, y));
    }
}
