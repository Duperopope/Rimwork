// Method to detect connected regions
private List<List<(int X, int Y)>> DetectRegions()
{
    var visited = new HashSet<(int X, int Y)>();
    var regions = new List<List<(int X, int Y)>>();

    for (int y = 0; y < Height; y++)
    {
        for (int x = 0; x < Width; x++)
        {
            if (!_grid[y, x].IsSolid && !visited.Contains((x, y)))
            {
                var region = new List<(int X, int Y)>();
                ExploreRegion(x, y, visited, region);
                regions.Add(region);
            }
        }
    }

    return regions;
}

private void ExploreRegion(int startX, int startY, HashSet<(int X, int Y)> visited, List<(int X, int Y)> region)
{
    var stack = new Stack<(int X, int Y)>();
    stack.Push((startX, startY));

    while (stack.Count > 0)
    {
        var (x, y) = stack.Pop();
        if (visited.Contains((x, y))) continue;
        visited.Add((x, y));
        region.Add((x, y));

        // Explore neighboring cells
        foreach (var (dx, dy) in new[] { (-1, 0), (1, 0), (0, -1), (0, 1) })
        {
            int nx = x + dx;
            int ny = y + dy;
            if (nx >= 0 && nx < Width && ny >= 0 && ny < Height && !_grid[ny, nx].IsSolid && !visited.Contains((nx, ny)))
            {
                stack.Push((nx, ny));
            }
        }
    }
}
