public bool IsZoneFunctional(Zone zone)
{
    // Check if the zone kind contributes to room functionality
    if (zone.Kind == ZoneKind.Canteen || zone.Kind == ZoneKind.SleepingQuarters)
    {
        // Check if there is at least one functional furniture in the zone
        var furnitureInZone = _furniture.Where(f => zone.Contains(f.X, f.Y)).ToList();
        return furnitureInZone.Any(f => f.IsFunctionalFurniture());
    }
    return false;
}
