public class Zone
{
    public ZoneKind Kind { get; }
    public int X { get; }
    public int Y { get; }
    public int Width { get; }
    public int Height { get; }
    public bool Contains(int x, int y) => x >= X && x < X + Width && y >= Y + Height;
    public (int X, int Y) CenterTile => (X + Width / 2, Y + Height / 2);

    public bool IsFunctional(GameMap map)
    {
        // Example: A zone is functional if it has at least one adjacent passable tile
        for (int dx = -1; dx <= 1; dx++)
        {
            for (int dy = -1; dy <= 1; dy++)
            {
                int adjX = X + dx;
                int adjY = Y + dy;
                if (map.IsPassable(adjX, adjY))
                {
                    return true;
                }
            }
        }
        return false;
    }
}
