private Dictionary<ResourceKind, int> _buildingMaterialConsumption = new();

public void TrackBuildingMaterialConsumption(ResourceKind kind, int amount)
{
    if (_buildingMaterialConsumption.ContainsKey(kind))
    {
        _buildingMaterialConsumption[kind] += amount;
    }
    else
    {
        _buildingMaterialConsumption[kind] = amount;
    }
}

public int GetTotalBuildingMaterialConsumption(ResourceKind kind)
{
    return _buildingMaterialConsumption.TryGetValue(kind, out int consumption) ? consumption : 0;
}
