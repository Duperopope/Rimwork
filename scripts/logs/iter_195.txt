public bool IsZoneFunctional(Zone zone)
{
    // Check if the zone kind contributes to room functionality
    if (zone.Kind == ZoneKind.Canteen || zone.Kind == ZoneKind.SleepingQuarters)
    {
        // Check if there is any functional furniture in the zone
        foreach (var furniture in _furniture)
        {
            if (furniture.IsFunctionalFurniture() && zone.Contains(furniture.X, furniture.Y))
            {
                return true;
            }
        }
    }
    return false;
}
