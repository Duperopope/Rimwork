public void AssignPawnsToTasks()
{
    foreach (var pawn in Pawns)
    {
        if (pawn.Needs.Food < 50)
        {
            var nearestFoodZone = FindNearestZoneTile(ZoneKind.Canteen, pawn.X, pawn.Y);
            if (nearestFoodZone.HasValue)
            {
                pawn.Tasks.AddTask(new Task(TaskKind.Eat, nearestFoodZone.Value));
            }
        }
        else if (pawn.Needs.Sleep < 50)
        {
            var nearestSleepZone = FindNearestZoneTile(ZoneKind.SleepingQuarters, pawn.X, pawn.Y);
            if (nearestSleepZone.HasValue)
            {
                pawn.Tasks.AddTask(new Task(TaskKind.Sleep, nearestSleepZone.Value));
            }
        }
    }
}
