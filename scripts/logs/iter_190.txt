public bool IsPlantable(int x, int y)
{
    if (!IsPassable(x, y)) return false;
    if (_resources.Exists(r => r.X == x && r.Y == y)) return false;
    if (_furniture.Exists(f => f.X == x && f.Y == y)) return false;
    if (_saplings.Exists(s => s.X == x && s.Y == y)) return false;
    return true;
}

private List<Room> DetectRooms()
{
    List<Room> rooms = new List<Room>();
    bool[,] visited = new bool[Height, Width];

    for (int y = 0; y < Height; y++)
    {
        for (int x = 0; x < Width; x++)
        {
            if (!visited[y, x] && IsPassable(x, y))
            {
                List<(int, int)> roomTiles = new List<(int, int)>();
                Queue<(int, int)> queue = new Queue<(int, int)>();
                queue.Enqueue((x, y));

                while (queue.Count > 0)
                {
                    var (currentX, currentY) = queue.Dequeue();
                    if (visited[currentY, currentX] || !IsPassable(currentX, currentY))
                        continue;

                    visited[currentY, currentX] = true;
                    roomTiles.Add((currentX, currentY));

                    for (int dy = -1; dy <= 1; dy++)
                    {
                        for (int dx = -1; dx <= 1; dx++)
                        {
                            if (dx == 0 && dy == 0) continue;
                            queue.Enqueue((currentX + dx, currentY + dy));
                        }
                    }
                }

                rooms.Add(new Room(roomTiles));
            }
        }
    }

    return rooms;
}
