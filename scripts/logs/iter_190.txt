public void SimulateResourceRegeneration()
{
    foreach (var resource in _recycledResources.Keys)
    {
        RegenerateResource(resource, _recycledResources[resource]);
    }

    LimitResourceConsumptionPerTick();
}

private void LimitResourceConsumptionPerTick()
{
    // Example: Limit wood consumption to 10 per tick
    int maxWoodConsumption = 10;
    if (Wood > maxWoodConsumption)
    {
        Wood = maxWoodConsumption;
    }

    // Example: Limit stone consumption to 5 per tick
    int maxStoneConsumption = 5;
    if (Stone > maxStoneConsumption)
    {
        Stone = maxStoneConsumption;
    }
}
