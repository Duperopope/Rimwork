public class GameMap
{
    private readonly Cell[,] _grid;
    private readonly List<ResourceNode> _resources = new();
    private readonly List<Furniture> _furniture = new();

    public bool RemoveResourceAt(int x, int y)
    {
        return _resources.RemoveAll(r => r.X == x && r.Y == y) > 0;
    }

    /// <summary>
    /// Detects contiguous regions of passable tiles and returns a list of rooms.
    /// </summary>
    public List<Room> DetectRooms()
    {
        var rooms = new List<Room>();
        bool[,] visited = new bool[_grid.GetLength(0), _grid.GetLength(1)];

        for (int y = 0; y < _grid.GetLength(0); y++)
        {
            for (int x = 0; x < _grid.GetLength(1); x++)
            {
                if (_grid[y, x].IsSolid || visited[y, x])
                    continue;

                var room = new Room();
                ExploreRoom(x, y, visited, room);
                rooms.Add(room);
            }
        }

        return rooms;
    }

    private void ExploreRoom(int x, int y, bool[,] visited, Room room)
    {
        if (x < 0 || x >= _grid.GetLength(1) || y < 0 || y >= _grid.GetLength(0) || _grid[y, x].IsSolid || visited[y, x])
            return;

        visited[y, x] = true;
        room.Tiles.Add((x, y));

        ExploreRoom(x + 1, y, visited, room);
        ExploreRoom(x - 1, y, visited, room);
        ExploreRoom(x, y + 1, visited, room);
        ExploreRoom(x, y - 1, visited, room);
    }
}
