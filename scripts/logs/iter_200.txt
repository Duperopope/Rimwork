public int CountFunctionalRooms()
{
    int functionalRoomCount = 0;
    foreach (var zone in _zones)
    {
        if (IsZoneFunctional(zone))
        {
            functionalRoomCount++;
        }
    }
    return functionalRoomCount;
}

public void ConsumeResourcesForWall()
{
    if (Wood >= WallWoodCost && Stone >= WallStoneCost)
    {
        Wood -= WallWoodCost;
        Stone -= WallStoneCost;
    }
    else
    {
        throw new InvalidOperationException("Insufficient resources to build a wall.");
    }
}
