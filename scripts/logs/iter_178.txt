public void TrackResourceConsumption(ResourceKind kind, int amount)
{
    if (_recycledResources.ContainsKey(kind))
    {
        _recycledResources[kind] -= amount;
        // Ensure the resource amount does not go below zero
        _recycledResources[kind] = Math.Max(_recycledResources[kind], 0);
    }
}
