public bool IsTilePartOfRoom(int x, int y)
{
    // Check if the tile is adjacent to a wall or furniture
    if (IsPassable(x, y))
    {
        return false;
    }

    // Check adjacent tiles
    for (int dx = -1; dx <= 1; dx++)
    {
        for (int dy = -1; dy <= 1; dy++)
        {
            int nx = x + dx;
            int ny = y + dy;
            if (nx >= 0 && nx < Width && ny >= 0 && ny < Height)
            {
                if (!IsPassable(nx, ny))
                {
                    return true;
                }
            }
        }
    }

    return false;
}
