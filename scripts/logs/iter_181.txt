public int GetResourceConsumption(Guid pawnId, ResourceKind kind)
{
    if (_pawnResourceConsumption.TryGetValue(pawnId, out var pawnConsumption))
    {
        pawnConsumption.TryGetValue(kind, out int amount);
        return amount;
    }
    return 0;
}

public int GetTotalResourceConsumption(ResourceKind kind)
{
    int totalConsumption = 0;

    foreach (var pawnConsumption in _pawnResourceConsumption.Values)
    {
        if (pawnConsumption.TryGetValue(kind, out int amount))
        {
            totalConsumption += amount;
        }
    }

    return totalConsumption;
