public void GenerateResourcesOverTime()
{
    // Example: Regenerate trees and rocks over time
    foreach (var resource in Resources)
    {
        if (resource.Kind == ResourceKind.Tree)
        {
            // Logic to regenerate trees
            // Example: Add a sapling if the tree is cut down
            if (!IsPlantable(resource.X, resource.Y))
            {
                AddSapling(resource.X, resource.Y, 100); // 100 ticks to grow
            }
        }
        else if (resource.Kind == ResourceKind.Rock)
        {
            // Logic to respawn rocks
            // Example: Respawn rocks every 500 ticks
            if (TotalTicks % 500 == 0)
            {
                _resources.Add(new ResourceNode(ResourceKind.Rock, resource.X, resource.Y));
            }
        }
    }
}
