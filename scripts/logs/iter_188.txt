public bool HasEnoughFunctionalFurniture(Zone zone, int requiredCount)
{
    if (zone == null)
    {
        return false;
    }

    var functionalFurnitureCount = zone.LocalMap._furniture.Count(furniture => furniture.IsFunctionalFurniture());
    return functionalFurnitureCount >= requiredCount;
}
