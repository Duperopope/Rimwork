private void RegisterPawn(string name)
{
	int num;
	int num2;
	do
	{
		num = _rng.Next(0, _world.Map.Width);
		num2 = _rng.Next(0, _world.Map.Height);
	}
	while (!_world.Map.IsPassable(num, num2));
	Pawn newPawn = new Pawn
	{
		Name = name,
		X = num,
		Y = num2,
		Sex = (PawnSex)_rng.Next(2), // Random sex
		SkillXP = new Dictionary<SkillKind, float>
		{
			{ SkillKind.Construction, _rng.Next(100) },
			{ SkillKind.Mining, _rng.Next(100) },
			{ SkillKind.Woodcutting, _rng.Next(100) }
		},
		Hunger = _rng.Next(100),
		Mood = _rng.Next(100),
		Fatigue = _rng.Next(100),
		HP = _rng.Next(100),
		Thirst = _rng.Next(100),
		Stress = _rng.Next(100)
	};
	_world.Registry.Add(newPawn.Id, newPawn);
}
