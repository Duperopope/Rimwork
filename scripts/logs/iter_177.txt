private Dictionary<ResourceKind, int> _resourceConsumption = new();

public void TrackResourceConsumption(ResourceKind kind, int amount)
{
    if (_resourceConsumption.ContainsKey(kind))
    {
        _resourceConsumption[kind] += amount;
    }
    else
    {
        _resourceConsumption[kind] = amount;
    }
}
