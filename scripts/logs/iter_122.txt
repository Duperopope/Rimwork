public void ToggleRain()
{
    _isRaining = !_isRaining;
    if (_isRaining)
    {
        // Reduce resource availability during rain
        foreach (var resource in _resources.ToList())
        {
            if (resource.Kind == ResourceKind.Tree)
            {
                RemoveResourceAt(resource.X, resource.Y);
            }
        }
    }
}
