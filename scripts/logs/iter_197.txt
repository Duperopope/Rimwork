public bool IsZoneFunctional(Zone zone)
{
    // Check if the zone kind is functional
    bool isFunctionalKind = zone.Kind == ZoneKind.Canteen || zone.Kind == ZoneKind.SleepingQuarters;

    // Check if the zone contains functional furniture
    bool hasFunctionalFurniture = zone.LocalMap.Furniture.Any(f => f.IsFunctionalFurniture());

    return isFunctionalKind && hasFunctionalFurniture;
}
