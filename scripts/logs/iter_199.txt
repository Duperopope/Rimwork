public bool HasEnoughFunctionalFurniture(Zone zone, int requiredCount)
{
    if (zone.LocalMap == null)
    {
        return false;
    }

    int functionalFurnitureCount = zone.LocalMap._furniture.Count(furniture => furniture.IsFunctionalFurniture() && zone.Contains(furniture.X, furniture.Y));
    return functionalFurnitureCount >= requiredCount;
}
