public bool IsZoneFunctional(Zone zone)
{
    switch (zone.Kind)
    {
        case ZoneKind.SleepingQuarters:
            return Furniture.Any(f => f.Kind == FurnitureKind.Bed && f.OwnerId.HasValue);
        case ZoneKind.Canteen:
            return Resources.Any(r => r.Kind == ResourceKind.Food);
        default:
            return false;
    }
}
