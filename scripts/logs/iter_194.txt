private void AssignPawnsToTasks()
{
    foreach (var pawn in Pawns)
    {
        if (pawn.Hunger > 50)
        {
            var foodZone = FindNearestZoneTile(ZoneKind.Canteen, pawn.X, pawn.Y);
            if (foodZone.HasValue)
            {
                pawn.Tasks.AddTask("Eat", foodZone.Value);
            }
        }
        else if (pawn.Fatigue > 50)
        {
            var sleepZone = FindNearestZoneTile(ZoneKind.SleepingQuarters, pawn.X, pawn.Y);
            if (sleepZone.HasValue)
            {
                pawn.Tasks.AddTask("Sleep", sleepZone.Value);
            }
        }
    }
}
