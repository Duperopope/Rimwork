public void SimulateResourceRegeneration()
{
    foreach (var resource in _recycledResources.Keys)
    {
        RegenerateResource(resource, _recycledResources[resource]);
    }

    // Track resource consumption for each pawn
    foreach (var pawn in Pawns)
    {
        foreach (var skill in pawn.SkillXP.Keys)
        {
            int consumption = GetSkillLevel(skill) * 2; // Example consumption logic
            ConsumeResource(ResourceKind.Wood, consumption);
        }
    }
}
