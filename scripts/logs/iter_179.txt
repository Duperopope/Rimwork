    public bool IsZoneFunctional(Zone zone)
    {
        // Check if the zone contains at least one functional furniture
        foreach (var furniture in Furniture)
        {
            if (furniture.IsFunctionalFurniture() && zone.Contains(furniture.X, furniture.Y))
            {
                return true;
            }
        }
        return false;
    }
