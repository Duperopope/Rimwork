public class Zone
{
    public ZoneKind Kind { get; }
    public int X { get; }
    public int Y { get; }
    public int Width { get; }
    public int Height { get; }

    public bool IsFunctional(GameMap map)
    {
        // A zone is functional if it has at least one passable tile
        for (int y = Y; y < Y + Height; y++)
        {
            for (int x = X; x < X + Width; x++)
            {
                if (map.IsPassable(x, y))
                {
                    return true;
                }
            }
        }
        return false;
    }
