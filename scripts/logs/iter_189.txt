{
	int num;
	int num2;
	do
	{
		num = _rng.Next(0, _world.Map.Width);
		num2 = _rng.Next(0, _world.Map.Height);
	}
	while (!_world.Map.IsPassable(num, num2));

	{

		Sex = (PawnSex)_rng.Next(2), // Random sex

		{
			{ SkillKind.Construction, 0f },
			{ SkillKind.Mining, 0f },
			{ SkillKind.Woodcutting, 0f }
		},
		Hunger = 0f,
		Mood = 0f,

		Relationships = new Dictionary<Guid, float>()
	};
	_world.Map.Registry[pawn.Id] = pawn;
	_world.Map.Pawns.Add(pawn);
