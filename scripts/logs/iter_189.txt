public bool IsPlantable(int x, int y)
{
    if (!IsPassable(x, y)) return false;
    if (_resources.Exists(r => r.X == x && r.Y == y)) return false;
    if (_furniture.Any(f => f.X == x && f.Y == y)) return false;
    if (_saplings.Any(s => s.X == x && s.Y == y)) return false;
    return true;
}

private bool IsFunctionalRoom(int x, int y, int width, int height)
{
    // Basic check: room must be enclosed by walls
    for (int i = x; i < x + width; i++)
    {
        if (!IsSolid(i, y) || !IsSolid(i, y + height - 1))
            return false;
    }
    for (int j = y; j < y + height; j++)
    {
        if (!IsSolid(x, j) || !IsSolid(x + width - 1, j))
            return false;
    }
    // Additional checks can be added here
    return true;
}
