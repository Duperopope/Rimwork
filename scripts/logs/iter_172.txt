        {

public bool IsZoneFunctional(Zone zone)
{
    // Check if the zone has at least one tile that is part of a room
    for (int y = zone.Y; y < zone.Y + zone.Height; y++)
    {
        for (int x = zone.X; x < zone.X + zone.Width; x++)
        {
            if (IsTilePartOfRoom(x, y))
            {
                return true;
            }
        }
    }
    return false;
}
