public void GenerateResourcesOverTime()
{
    // Generate resources based on biome
    foreach (var region in _regions)
    {
        for (int x = region.X; x < region.X + 10; x++)
        {
            for (int y = region.Y; y < region.Y + 10; y++)
            {
                if (IsPlantable(x, y) && !_noGrowTiles.Contains((x, y)))
                {
                    switch (region.Biome)
                    {
                        case "Forest":
                            AddResource(ResourceKind.Tree, x, y);
                            break;
                        case "Desert":
                            AddResource(ResourceKind.Rock, x, y);
                            break;
                        // Add more biomes as needed
                    }
                }
            }
        }
    }
}

private void AddResource(ResourceKind kind, int x, int y)
{
    _resources.Add(new ResourceNode(kind, x, y));
