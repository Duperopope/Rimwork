public void AssignPawnsToTasksBasedOnSleepNeed()
{
    foreach (var pawn in Pawns)
    {
        if (pawn.Needs.Mood < 50) // Example threshold for sleep need
        {
            var sleepZone = Zones.FirstOrDefault(z => z.Kind == ZoneKind.SleepingQuarters);
            if (sleepZone != null)
            {
                var task = new TaskBoard.Task
                {
                    PawnId = pawn.Id,
                    Kind = TaskBoard.TaskKind.GoToSleep,
                    Target = sleepZone.CenterTile
                };
                Tasks.Add(task);
            }
        }
    }
}
