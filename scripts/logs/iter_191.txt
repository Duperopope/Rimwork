public bool IsPlantable(int x, int y)
{
    if (!IsPassable(x, y)) return false;
    if (_resources.Exists(r => r.X == x && r.Y == y)) return false;

    // Check for existing furniture or saplings
    if (_furniture.Any(f => f.X == x && f.Y == y)) return false;
    if (_saplings.Any(s => s.X == x && s.Y == y)) return false;

    return true;
}

// Method to detect rooms
public List<Room> GetRooms()
{
    List<Room> rooms = new List<Room>();
    bool[,] visited = new bool[Height, Width];

    for (int y = 0; y < Height; y++)
    {
        for (int x = 0; x < Width; x++)
        {
            if (!visited[y, x] && IsPassable(x, y))
            {
                Room room = new Room();
                ExploreRoom(x, y, visited, room);
                rooms.Add(room);
            }
        }
    }

    return rooms;
}

private void ExploreRoom(int startX, int startY, bool[,] visited, Room room)
{
    Stack<(int X, int Y)> stack = new Stack<(int X, int Y)>();
    stack.Push((startX, startY));
    visited[startY, startX] = true;

    while (stack.Count > 0)
    {
        var (x, y) = stack.Pop();
        room.Tiles.Add((x, y));

        // Check adjacent tiles
        int[] dx = { -1, 1, 0, 0 };
        int[] dy = { 0, 0, -1, 1 };
        for (int i = 0; i < 4; i++)
        {
            int newX = x + dx[i];
            int newY = y + dy[i];
            if (newX >= 0 && newX < Width && newY >= 0 && newY < Height &&
                !visited[newY, newX] && IsPassable(newX, newY))
            {
                stack.Push((newX, newY));
                visited[newY, newX] = true;
            }
        }
    }
}
