public bool HasEnoughFunctionalFurniture(Zone zone, int requiredCount)
{
    if (zone == null)
    {
        return false;
    }

    int functionalFurnitureCount = zone.LocalMap.Furniture.Count(furniture => furniture.IsFunctionalFurniture());
    return functionalFurnitureCount >= requiredCount;
}
