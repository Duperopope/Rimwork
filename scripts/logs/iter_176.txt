public void SimulateResourceRegeneration()
{
    foreach (var resource in _recycledResources.Keys)
    {
        RegenerateResource(resource, _recycledResources[resource]);
    }

    LimitResourceUsage();
}

private void LimitResourceUsage()
{
    // Example: Limit wood usage to 50 units
    if (_recycledResources.TryGetValue(ResourceKind.Tree, out int woodAmount))
    {
        _recycledResources[ResourceKind.Tree] = Math.Min(woodAmount, 50);
    }

    // Example: Limit stone usage to 30 units
    if (_recycledResources.TryGetValue(ResourceKind.Rock, out int stoneAmount))
    {
        _recycledResources[ResourceKind.Rock] = Math.Min(stoneAmount, 30);
    }
}
