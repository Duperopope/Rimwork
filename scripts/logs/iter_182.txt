public bool CanUnlockBuildingType(ResourceKind kind, int requiredAmount)
{
    int totalConsumption = GetTotalResourceConsumption(kind);
    int totalRecycled = _recycledResources.ContainsKey(kind) ? _recycledResources[kind] : 0;
    int totalAvailable = totalConsumption + totalRecycled;

    return totalAvailable >= requiredAmount;
}
